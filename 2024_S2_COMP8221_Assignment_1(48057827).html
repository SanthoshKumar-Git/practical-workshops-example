import torch
import torch.nn as nn

# Defining the PixelCNN class which is a subclass of nn.Module
class PixelCNN(nn.Module):
    def __init__(self, in_channels=3, num_classes=256, num_layers=12, kernel_size=3):
        super(PixelCNN, self).__init__()  # Calling the superclass constructor (nn.Module)
        
        # Defining a list of convolutional layers (nn.Conv2d) as part of the model
        # The first layer takes input channels (e.g., RGB: 3 channels) and outputs 'num_classes' channels
        # Subsequent layers have 'num_classes' as both input and output channels
        # The kernel size is 3x3 for all layers, and padding=1 ensures spatial dimensions remain the same
        self.layers = nn.ModuleList([
            nn.Conv2d(in_channels, num_classes, kernel_size, padding=1)  # Defining the first convolutional layer
            if i == 0 else  # Checking if it's the first layer
            nn.Conv2d(num_classes, num_classes, kernel_size, padding=1)  # Defining subsequent convolutional layers
            for i in range(num_layers)  # Creating 'num_layers' convolutional layers
        ])
        
        # Defining the output layer, a 1x1 convolutional layer, to map the feature map to output classes
        self.output_layer = nn.Conv2d(num_classes, num_classes, kernel_size=1)

    # Defining the forward pass of the model
    def forward(self, x):
        out = x  # Assigning the input tensor to 'out'
        
        # Passing the input through each convolutional layer in the list, applying ReLU activation after each
        for layer in self.layers:
            out = torch.relu(layer(out))  # Applying ReLU non-linearity after each convolutional layer
        
        # Applying the output layer (1x1 convolution) after passing through all previous layers
        out = self.output_layer(out)
        
        # Returning the final output tensor
        return out

# Instantiating the PixelCNN model
model = PixelCNN()

# Printing the architecture of the PixelCNN model
print("PixelCNN Model Architecture:")
print(model)

import torch.optim as optim
from tqdm import tqdm

# Initializing the PixelCNN model
model = PixelCNN()

# Setting the device to GPU if available, otherwise falling back to CPU
device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
model.to(device)  # Moving the model to the specified device

# Retrieving model parameters to be optimized
params = list(model.parameters())
if len(params) == 0:
    # Raising an error if no parameters are found in the model
    raise ValueError("Model has no parameters to optimize. Check the model architecture.")

# Initializing the Adam optimizer with model parameters and a learning rate of 0.001
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Printing statements for verifying model, optimizer, and hyperparameters
print("Model, optimizer, and hyperparameters initialized.")
print(optimizer)

# Setting the number of epochs for training
num_epochs = 10

# Beginning the training loop for each epoch
for epoch in range(num_epochs):
    model.train()  # Setting the model to training mode
    running_loss = 0.0  # Initializing running loss for each epoch

    # Iterating through batches of training data
    for images, _ in tqdm(train_loader, desc=f"Epoch {epoch+1}/{num_epochs}", unit="batch"):
        images = images.to(device)  # Moving the input images to the device (GPU/CPU)

        optimizer.zero_grad()  # Zeroing the gradients for the optimizer

        # Forward pass: Getting model predictions
        outputs = model(images)

        # Computing the loss using cross-entropy between the predicted and target labels
        loss = criterion(outputs, images.argmax(dim=1))

        # Backpropagating the loss
        loss.backward()

        # Updating the model parameters
        optimizer.step()

        # Updating running loss by multiplying with the number of images in the batch
        running_loss += loss.item() * images.size(0)

    # Calculating the average loss per epoch
    epoch_loss = running_loss / len(train_loader.dataset)

    # Printing the loss for the current epoch
    print(f"Epoch {epoch+1}/{num_epochs}, Loss: {epoch_loss:.4f}")

def calculate_log_likelihood(model, data_loader, device):
    # Setting the model to evaluation mode to disable certain layers like dropout
    model.eval()
    # Initializing total log-likelihood accumulator
    total_log_likelihood = 0.0

    # Disabling gradient computation as it's not required during evaluation
    with torch.no_grad():
        # Looping through batches of data in the data_loader
        for images, _ in data_loader:
            # Moving the images to the specified device (GPU/CPU)
            images = images.to(device)

            # Running the model on the input images to get the output predictions
            outputs = model(images)

            # Calculating the negative log-likelihood using the defined loss criterion
            log_likelihood = -criterion(outputs, images.argmax(dim=1)).mean()

            # Accumulating the log-likelihood, scaled by the batch size
            total_log_likelihood += log_likelihood.item() * images.size(0)

    # Returning the average log-likelihood over the entire dataset
    return total_log_likelihood / len(data_loader.dataset)

import torch

# Defining the MultiScalePixelCNN class
class MultiScalePixelCNN(nn.Module):
    def __init__(self, in_channels=3, num_classes=256, num_layers=12, kernel_size=3):
        super(MultiScalePixelCNN, self).__init__()

        # Initializing three branches of PixelCNN
        self.branch1 = PixelCNN(in_channels, num_classes, num_layers, kernel_size)
        self.branch2 = PixelCNN(in_channels, num_classes, num_layers, kernel_size)
        self.branch3 = PixelCNN(in_channels, num_classes, num_layers, kernel_size)

    def forward(self, x):
        # Passing input through the first branch
        out1 = self.branch1(x)
        # Passing output of the first branch through the second branch
        out2 = self.branch2(out1)
        # Passing output of the second branch through the third branch
        out3 = self.branch3(out2)
        # Returning final output
        return out3

import torch
import matplotlib.pyplot as plt

# Defining the function to generate samples from the model
def generate_samples(model, num_samples, device):
    model.eval()  # Set the model to evaluation mode
    # Initialize a tensor to hold the generated samples
    samples = torch.zeros(num_samples, 3, 32, 32).to(device)
    with torch.no_grad():  # Disable gradient calculation
        # Iterate over each pixel location in the 32x32 grid
        for i in range(32):
            for j in range(32):
                for c in range(3):
                    # Generate model output for current state of samples
                    output = model(samples)
                    # Compute probabilities for the current pixel location
                    probs = torch.softmax(output[:, :, i, j], dim=1)
                    # Sample from the multinomial distribution based on the computed probabilities
                    samples[:, c, i, j] = torch.multinomial(probs, 1).squeeze(-1)
    return samples

# Defining the number of samples to generate

# Show the plot

Training involved processing images in batches, computing the loss, performing backpropagation, and updating model parameters over the course of 10 epochs.


Evaluation:

Results:
The PixelCNN model achieved a test log-likelihood of approximately -1.059, indicating reasonable performance but suggesting potential for further optimization.


Sample Generation:
